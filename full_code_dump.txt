# ============================================================
# src/data/load_data.py
#
# Paper V-A-1: 3 dataset — PaySim, Credit Card 2023, IEEE-CIS.
#   IEEE-CIS: merge transaction + identity bằng TransactionID,
#             "handling cases where identity information is missing"
#             -> LEFT JOIN, giữ NaN (sẽ được impute ở bước preprocess).
#
# QUY TẮC: không tạo/giả lập dữ liệu. File thiếu -> báo lỗi rõ ràng.
# [INFERRED] sample_frac chỉ để chạy thử nhanh (KHÔNG phải paper); mặc định null.
# ============================================================
from __future__ import annotations

from pathlib import Path
from typing import Any, Dict

import pandas as pd


def _need(path: Path, hint: str) -> None:
    if not path.exists():
        raise FileNotFoundError(
            f"Không tìm thấy file dữ liệu: {path}\n"
            f"  -> {hint}\n"
            f"  (Project này KHÔNG tự sinh dữ liệu giả; hãy đặt file thật vào data/raw/.)"
        )


def load_dataset(cfg: Dict[str, Any]) -> pd.DataFrame:
    ds = cfg["dataset"]
    label_col = ds["label_col"]

    if "transaction_path" in ds:  # IEEE-CIS gốc (2 file)
        tx_path = Path(ds["transaction_path"])
        _need(tx_path, "Tải train_transaction.csv từ Kaggle ieee-fraud-detection.")
        print(f"[LOAD] Reading transactions: {tx_path}")
        df = pd.read_csv(tx_path)
        id_p = ds.get("identity_path")
        if id_p and Path(id_p).exists():
            print(f"[LOAD] Reading identity: {id_p}")
            ident = pd.read_csv(id_p)
            key = ds.get("key_col", "TransactionID")
            df = df.merge(ident, how="left", on=key)  # giữ NaN khi thiếu identity
            print(f"[LOAD] Merged on {key}: -> {df.shape}")
        else:
            print("[LOAD] [WARN] Không có identity file -> chỉ dùng transaction features")
    else:
        path = Path(ds["path"])
        _need(path, "Đặt CSV (PaySim / CC2023 / IEEE-CIS đã merge) vào đường dẫn trong config.")
        print(f"[LOAD] Reading: {path}")

    if label_col not in df.columns:

==================== src/data/preprocess.py ====================
# ============================================================
# src/data/preprocess.py — Paper V-A-1-d (nguyên văn):
#   - Missing value imputation: MEAN (numeric), MODE (categorical)
#   - Feature scaling: MIN-MAX (numeric)
#   - One-hot encoding (categorical)
#   - Temporal features: hour of day, day of week, time differences
#     between consecutive transactions
#   - IEEE-CIS: merge transaction + identity (xử lý ở load_data.py)
#
# fit() CHỈ trên train; transform() cho val/test (không rò rỉ).
#
# [INFERRED] chi tiết paper không nói:
#   - hour/day_of_week/delta_t tính theo đơn vị dataset.time_unit (second|hour),
#     delta_t tính trong từng split sau khi sort theo thời gian.
#   - Temporal features được đưa vào như numeric (min-max) — bản cũ đã loại nhầm.
# ============================================================
from __future__ import annotations

import warnings
from typing import Any, Dict, List, Tuple

import joblib
import numpy as np
import pandas as pd
from scipy import sparse
from sklearn.compose import ColumnTransformer
from sklearn.impute import SimpleImputer
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import MinMaxScaler, OneHotEncoder

TEMPORAL_COLS = ["__hour_of_day", "__day_of_week", "__delta_t"]


class FraudPreprocessor:
    def __init__(self, cfg: Dict[str, Any]):
        ds, pre = cfg.get("dataset", {}), cfg.get("preprocess", {})
        self.label_col = ds.get("label_col", "isFraud")
        self.time_col = ds.get("time_col")
        self.time_unit = ds.get("time_unit", "second")
        self.numeric_strategy = pre.get("numeric_strategy", "mean")
        self.categorical_strategy = pre.get("categorical_strategy", "most_frequent")
        self.drop_cols = list(pre.get("drop_cols", []))
        self.categorical_cols_cfg = list(pre.get("categorical_cols", []))
        self.onehot_min_frequency = pre.get("onehot_min_frequency")  # None = đúng paper
        self.numeric_cols: List[str] = []
        self.categorical_cols: List[str] = []
        self.preprocessor: ColumnTransformer | None = None

    # ------------------------------------------------------------------
    def _add_temporal_features(self, df: pd.DataFrame) -> pd.DataFrame:
        if not self.time_col or self.time_col not in df.columns:
            return df
        df = df.copy()
        t = pd.to_numeric(df[self.time_col], errors="coerce").fillna(0).astype(float).to_numpy()
        order = np.argsort(t, kind="stable")
        ts = t[order]
        hours = ts / 3600.0 if self.time_unit == "second" else ts
        hour = np.floor(hours % 24)
        dow = np.floor(hours / 24) % 7
        dt = np.diff(ts, prepend=ts[:1])
        for name, arr in zip(TEMPORAL_COLS, (hour, dow, dt)):
            full = np.empty(len(df), dtype=np.float32)
            full[order] = arr
            df[name] = full
        return df

    @staticmethod
    def _clean_categorical(df: pd.DataFrame, cols: List[str]) -> pd.DataFrame:
        """Chuỗi hoá giá trị nhưng GIỮ NaN để SimpleImputer(most_frequent) xử lý đúng (mode)."""
        for c in cols:
            if c in df.columns:
                s = df[c]
                df[c] = s.where(s.isna(), s.astype(str)).astype(object)
        return df

    def _identify_columns(self, df: pd.DataFrame) -> None:
        drop = {self.label_col, *self.drop_cols}
        if self.time_col:
            drop.add(self.time_col)
        feats = [c for c in df.columns if c not in drop]
        explicit_cat = set(self.categorical_cols_cfg)
        num, cat = [], []
        for c in feats:
            if c in explicit_cat or not pd.api.types.is_numeric_dtype(df[c]):
                cat.append(c)
            else:
                num.append(c)
        self.numeric_cols, self.categorical_cols = sorted(num), sorted(cat)
        if len(num) + len(cat) > 1000:
            warnings.warn(f"{len(num)+len(cat)} cột — có thể tràn RAM sau one-hot (xem preprocess.onehot_min_frequency).")

    def _build(self) -> ColumnTransformer:
        tf = []
        if self.numeric_cols:
            tf.append(("num", Pipeline([
                ("imputer", SimpleImputer(strategy=self.numeric_strategy)),
                ("scaler", MinMaxScaler()),
            ]), self.numeric_cols))
        if self.categorical_cols:
            tf.append(("cat", Pipeline([
                ("imputer", SimpleImputer(strategy=self.categorical_strategy)),
                ("onehot", OneHotEncoder(handle_unknown="infrequent_if_exist" if self.onehot_min_frequency else "ignore",
                                         min_frequency=self.onehot_min_frequency,
                                         sparse_output=True, dtype=np.float32)),
            ]), self.categorical_cols))
        return ColumnTransformer(tf, remainder="drop", verbose_feature_names_out=False)

    # ------------------------------------------------------------------
    def fit(self, df: pd.DataFrame) -> "FraudPreprocessor":
        df = self._add_temporal_features(df)
        self._identify_columns(df)
        df = self._clean_categorical(df.copy(), self.categorical_cols)
        self.preprocessor = self._build().fit(df)
        return self

    def transform(self, df: pd.DataFrame) -> Tuple[np.ndarray, np.ndarray, np.ndarray | None]:
        if self.preprocessor is None:
            raise RuntimeError("FraudPreprocessor chưa fit().")
        df = self._add_temporal_features(df)
        df = self._clean_categorical(df.copy(), self.categorical_cols)
        y = df[self.label_col].astype(int).to_numpy()
        t = (pd.to_numeric(df[self.time_col], errors="coerce").fillna(0).astype(float).to_numpy()
             if self.time_col and self.time_col in df.columns else None)
        x = self.preprocessor.transform(df)
        if sparse.issparse(x):
            x = x.toarray()
        return np.asarray(x, dtype=np.float32), y, t

    def fit_transform(self, df: pd.DataFrame):
        return self.fit(df).transform(df)

    def get_feature_names_out(self) -> List[str]:
        return list(self.preprocessor.get_feature_names_out()) if self.preprocessor else []

    def save(self, path: str) -> None:
        joblib.dump(self, path)


def load_preprocessor(path: str) -> FraudPreprocessor:
    return joblib.load(path)


__all__ = ["FraudPreprocessor", "load_preprocessor", "TEMPORAL_COLS"]

# ============================================================
# src/data/split.py
#
# Paper V-A-1-c: "we maintain the original train-test splits where provided,
# and for PaySim, we use an 80-20 random split while preserving the temporal
# order of transactions."
#
# [INFERRED] Diễn giải thống nhất cho cả 3 dataset (80/20 ngoài, val lấy từ 80%):
#   train 0.64 / val 0.16 / test 0.20
#   - temporal           : cắt theo time_col (PaySim, IEEE-CIS) — "random" của
#                          paper mâu thuẫn với "preserving temporal order";
#                          chọn temporal để không rò rỉ tương lai.
#   - stratified_random  : CC2023 (không có cột thời gian)
#   IEEE-CIS: Kaggle không công bố nhãn test -> chỉ có train_transaction -> temporal.
#
# CV 5-fold (V-A-4): temporal -> expanding-window TimeSeriesSplit;
#                    không có thời gian -> StratifiedKFold.
# ============================================================
from __future__ import annotations

from typing import Any, Dict, Iterator, Tuple

import numpy as np
import pandas as pd
from sklearn.model_selection import StratifiedKFold, TimeSeriesSplit, train_test_split


def _validate_split(name: str, df: pd.DataFrame, label_col: str) -> None:
    if len(df) == 0:
        raise ValueError(f"{name} split rỗng.")
    m = df[label_col].mean()
    if m in (0.0, 1.0):
        raise ValueError(f"{name} split chỉ có một lớp (fraud rate={m:.0%}). Kiểm tra strategy/time_col.")


def _temporal_cut(df: pd.DataFrame, time_col: str, train_ratio: float, val_ratio: float):
    df = df.sort_values(time_col, kind="stable").reset_index(drop=True)
    n = len(df)
    a, b = int(n * train_ratio), int(n * (train_ratio + val_ratio))
    return df.iloc[:a].copy(), df.iloc[a:b].copy(), df.iloc[b:].copy()


def split_dataframe(df: pd.DataFrame, cfg: Dict[str, Any]) -> Tuple[pd.DataFrame, pd.DataFrame, pd.DataFrame]:
    ds, sp = cfg["dataset"], cfg.get("split", {})
    label_col, time_col = ds["label_col"], ds.get("time_col")
    strategy = sp.get("strategy", "temporal")
    tr, va = float(sp.get("train_ratio", 0.64)), float(sp.get("val_ratio", 0.16))
    seed = int(ds.get("random_state", 42))
    if not (0 < tr < 1 and 0 < va < 1 and tr + va < 1):
        raise ValueError("train_ratio/val_ratio không hợp lệ.")

    df = df.reset_index(drop=True)
    if strategy == "temporal":
        if not time_col:
            raise ValueError("strategy='temporal' cần dataset.time_col.")
        train_df, val_df, test_df = _temporal_cut(df, time_col, tr, va)
    elif strategy in ("stratified_random", "random"):
        strat = df[label_col] if strategy == "stratified_random" and df[label_col].nunique() > 1 else None
        train_df, rest = train_test_split(df, train_size=tr, random_state=seed, stratify=strat)
        s2 = rest[label_col] if strat is not None else None
        val_df, test_df = train_test_split(rest, train_size=va / (1 - tr), random_state=seed, stratify=s2)
    else:
        raise ValueError(f"Unknown split strategy: {strategy}")

    for name, part in (("Train", train_df), ("Val", val_df), ("Test", test_df)):
        _validate_split(name, part, label_col)

    print(f"[SPLIT] {strategy}: train={len(train_df):,} ({train_df[label_col].mean()*100:.4f}% fraud) | "
          f"val={len(val_df):,} ({val_df[label_col].mean()*100:.4f}%) | "
          f"test={len(test_df):,} ({test_df[label_col].mean()*100:.4f}%)")
    return tuple(p.reset_index(drop=True) for p in (train_df, val_df, test_df))  # type: ignore


def get_cv_splits(df: pd.DataFrame, cfg: Dict[str, Any], n_folds: int = 5
                  ) -> Iterator[Tuple[pd.DataFrame, pd.DataFrame, pd.DataFrame]]:
    """Sinh (train_df, val_df, test_df) cho mỗi fold. Val = phần cuối/ngẫu nhiên 20% của train-fold."""
    ds, sp = cfg["dataset"], cfg.get("split", {})
    label_col, time_col = ds["label_col"], ds.get("time_col")
    seed = int(ds.get("random_state", 42))
    val_frac = float(sp.get("cv_val_frac", 0.2))
    df = df.reset_index(drop=True)

    if sp.get("strategy", "temporal") == "temporal" and time_col:
        df = df.sort_values(time_col, kind="stable").reset_index(drop=True)
        print(f"[CV] TimeSeriesSplit({n_folds}) — expanding window, giữ thứ tự thời gian")
        for tr_idx, te_idx in TimeSeriesSplit(n_splits=n_folds).split(df):
            n_va = max(1, int(len(tr_idx) * val_frac))
            yield (df.iloc[tr_idx[:-n_va]].copy().reset_index(drop=True),
                   df.iloc[tr_idx[-n_va:]].copy().reset_index(drop=True),
                   df.iloc[te_idx].copy().reset_index(drop=True))
    else:
        print(f"[CV] StratifiedKFold({n_folds})")
        skf = StratifiedKFold(n_splits=n_folds, shuffle=True, random_state=seed)
        for tr_idx, te_idx in skf.split(df, df[label_col]):
            tr_df = df.iloc[tr_idx]
            train_df, val_df = train_test_split(tr_df, test_size=val_frac, random_state=seed,
                                                stratify=tr_df[label_col])
            yield (train_df.reset_index(drop=True), val_df.reset_index(drop=True),
                   df.iloc[te_idx].copy().reset_index(drop=True))


def resample_fraud_ratio(df: pd.DataFrame, ratio_pct: float, label_col: str,
                         time_col: str | None, seed: int = 42) -> pd.DataFrame:
    """
    Thí nghiệm class-imbalance (V-B-3, Fig. 4): ép tỉ lệ fraud về ratio_pct (%) bằng
    random sampling và GIỮ thứ tự thời gian.
      - ratio < hiện tại : giữ toàn bộ normal, bớt fraud.
      - ratio > hiện tại : giữ toàn bộ fraud, bớt normal.
    [INFERRED] Paper chỉ nói "random sampling preserving temporal order".
    """
    r = ratio_pct / 100.0
    rng = np.random.default_rng(seed)
    pos, neg = df[df[label_col] == 1], df[df[label_col] == 0]
    cur = len(pos) / max(1, len(df))
    if r >= cur:
        n_neg = int(round(len(pos) * (1 - r) / r))
        neg = neg.iloc[np.sort(rng.choice(len(neg), size=min(n_neg, len(neg)), replace=False))]
    else:
        n_pos = int(round(len(neg) * r / (1 - r)))
        pos = pos.iloc[np.sort(rng.choice(len(pos), size=min(n_pos, len(pos)), replace=False))]
    out = pd.concat([pos, neg])
    out = out.sort_values(time_col, kind="stable") if time_col else out.sort_index()
    return out.reset_index(drop=True)

# ============================================================
# src/eval/adversarial.py — Robustness trước tấn công đối kháng.
#
# Paper: abstract nêu "resilience to ... adversarial attacks"; Future Work (5) "enhancing adversarial
# robustness". Paper KHÔNG mô tả giao thức tấn công -> [INFERRED] 3 tấn công TEST-TIME chuẩn:
#   - fgsm            : x' = clip(x + eps * sign(grad_x BCE), 0, 1)   (white-box, đặc trưng đã min-max [0,1])
#   - random_noise    : x' = clip(x + N(0, eps^2), 0, 1)
#   - edge_perturb    : thêm/xoá ngẫu nhiên tỉ lệ `eps` cạnh (tấn công cấu trúc đồ thị)
# (Bản cũ "label flipping" lật điểm số đã dự đoán nên không phải tấn công thật — đã loại.)
# Ngưỡng quyết định giữ NGUYÊN threshold của pipeline để so sánh công bằng.
# ============================================================
from __future__ import annotations

import json
from pathlib import Path
from typing import Dict, List, Optional

import numpy as np
import torch
import torch.nn.functional as F

from .evaluate import predict_scores
from .metrics import classification_metrics


def _fgsm(model, data, device, eps):
    model.eval()
    d = data.clone().to(device)
    d.x = d.x.clone().requires_grad_(True)
    loss = F.binary_cross_entropy_with_logits(model(d), d.y.float())
    grad, = torch.autograd.grad(loss, d.x)
    d.x = (d.x.detach() + eps * grad.sign()).clamp(0.0, 1.0)
    return d


def _noise(model, data, device, eps):
    d = data.clone().to(device)
    d.x = (d.x + torch.randn_like(d.x) * eps).clamp(0.0, 1.0)
    return d


def _edge_perturb(model, data, device, eps):
    d = data.clone().to(device)
    E, N = d.edge_index.size(1), d.num_nodes
    if E == 0:
        return d
    keep = torch.rand(E, device=device) >= eps                      # xoá eps cạnh
    n_add = int(E * eps)
    add = torch.randint(0, N, (2, n_add), device=device)             # thêm eps cạnh ngẫu nhiên
    has_dt = d.edge_time_delta.numel() == E
    d.edge_index = torch.cat([d.edge_index[:, keep], add], dim=1)
    if has_dt:
        d.edge_time_delta = torch.cat([d.edge_time_delta[keep], torch.zeros(n_add, device=device)])
    return d


ATTACKS = {"fgsm": _fgsm, "random_noise": _noise, "edge_perturb": _edge_perturb}


def evaluate_adversarial_robustness(model, test_graph, threshold: float = 0.5, device: str = "cpu",
                                    attacks: Optional[List[str]] = None, epsilon: float = 0.1,
                                    seed: int = 42, output_dir: Optional[str] = None,
                                    eval_chunk: Optional[int] = None) -> Dict:
    attacks = attacks or list(ATTACKS)
    torch.manual_seed(seed); np.random.seed(seed)
    model.to(device).eval()
    s0, y = predict_scores(model, test_graph, device, chunk_nodes=eval_chunk)
    base = classification_metrics(y, s0, threshold)
    res: Dict = {"epsilon": epsilon, "threshold": threshold,
                 "clean": {k: base[k] for k in ("auc_roc", "auc_pr", "f1", "recall", "precision", "fpr", "recall_at_1pct")}}
    for name in attacks:
        try:
            adv = ATTACKS[name](model, test_graph, device, epsilon)
            s, _ = predict_scores(model, adv, device, chunk_nodes=eval_chunk)
            m = classification_metrics(y, s, threshold)
            res[name] = {k: m[k] for k in ("auc_roc", "auc_pr", "f1", "recall", "precision", "fpr", "recall_at_1pct")}
            for k in ("f1", "auc_roc", "auc_pr", "recall"):
                res[name][f"{k}_drop_pct"] = float((base[k] - m[k]) / max(base[k], 1e-8) * 100)
        except Exception as e:  # noqa
            res[name] = {"error": repr(e)}
    if output_dir:
        p = Path(output_dir); p.mkdir(parents=True, exist_ok=True)
        (p / "adversarial_results.json").write_text(json.dumps(res, indent=2), encoding="utf-8")
    return res


run_adversarial_test = evaluate_adversarial_robustness

# ============================================================
# src/eval/concept_drift.py — Độ bền trước concept drift (Paper abstract / Section I).
#
# Paper nêu "maintaining high performance over extended periods" nhưng KHÔNG mô tả giao thức.
# [INFERRED]: sắp xếp tập test theo thời gian, chia thành `n_windows` cửa sổ bằng số mẫu, tính metric
# từng cửa sổ với CÙNG threshold; báo cáo suy giảm tương đối (cửa sổ đầu -> cuối) và độ dốc tuyến tính.
# Cửa sổ không có fraud: AUC không xác định -> bỏ qua (ghi `skipped_windows`).
# ============================================================
from __future__ import annotations

import json
from pathlib import Path
from typing import Dict, Optional

import numpy as np

from .metrics import classification_metrics


def evaluate_concept_drift(scores, labels, threshold: float = 0.5, n_windows: int = 5,
                           y_pred=None) -> Dict:
    scores, labels = np.asarray(scores), np.asarray(labels)
    n = len(scores)
    edges = np.linspace(0, n, n_windows + 1).astype(int)
    wins, skipped = [], 0
    for i in range(n_windows):
        a, b = edges[i], edges[i + 1]
        if b - a == 0 or labels[a:b].sum() == 0 or labels[a:b].sum() == b - a:
            skipped += 1
            continue
        m = classification_metrics(labels[a:b], scores[a:b], threshold, None if y_pred is None else np.asarray(y_pred)[a:b])
        wins.append({"window": i, "n": int(b - a), "fraud_rate": float(labels[a:b].mean()),
                     **{k: m[k] for k in ("auc_roc", "auc_pr", "f1", "recall", "precision", "fpr", "recall_at_1pct")}})
    out: Dict = {"n_windows": n_windows, "skipped_windows": skipped, "windows": wins}
    if len(wins) >= 2:
        for k in ("auc_roc", "auc_pr", "f1", "recall_at_1pct"):
            v = [w[k] for w in wins]
            out[f"{k}_decay_pct"] = float((v[0] - v[-1]) / max(v[0], 1e-8) * 100)
            out[f"{k}_slope"] = float(np.polyfit(np.arange(len(v)), v, 1)[0])
    return out


def run_concept_drift_test(scores, labels, threshold: float = 0.5, n_windows: int = 5,
                           y_pred=None, output_dir: Optional[str] = None) -> Dict:
    res = evaluate_concept_drift(scores, labels, threshold, n_windows, y_pred)
    if output_dir:
        p = Path(output_dir); p.mkdir(parents=True, exist_ok=True)
        (p / "concept_drift_results.json").write_text(json.dumps(res, indent=2), encoding="utf-8")
    return res

# ============================================================
# src/eval/evaluate.py — suy luận + lưu kết quả.
# ============================================================
from __future__ import annotations

import json
import time
from pathlib import Path
from typing import Dict, Optional, Tuple

import numpy as np
import torch


def iter_node_chunks(n: int, chunk: int, order: Optional[torch.Tensor] = None, shuffle: bool = False):
    idx = order if order is not None else torch.arange(n)
    if shuffle:
        starts = list(range(0, n, chunk))
        np.random.shuffle(starts)
    else:
        starts = range(0, n, chunk)
    for a in starts:
        yield idx[a:a + chunk]


@torch.no_grad()
def predict_scores(model, data, device: str = "cpu", timing: Optional[Dict[str, float]] = None,
                   chunk_nodes: Optional[int] = None) -> Tuple[np.ndarray, np.ndarray]:
    """Trả (scores, labels). chunk_nodes=None: suy luận toàn đồ thị (đúng paper)."""
    model.eval()
    t0 = time.perf_counter()
    if chunk_nodes is None or data.num_nodes <= chunk_nodes:
        scores = torch.sigmoid(model(data.to(device))).cpu().numpy().ravel()
    else:  # cạnh xuyên chunk bị bỏ — chỉ dùng khi hết bộ nhớ
        out = np.zeros(data.num_nodes, dtype=np.float32)
        for idx in iter_node_chunks(data.num_nodes, chunk_nodes):
            sub = data.subgraph(idx).to(device)
            out[idx.numpy()] = torch.sigmoid(model(sub)).cpu().numpy().ravel()
        scores = out
    if timing is not None:
        timing["inference_sec"] = timing.get("inference_sec", 0.0) + time.perf_counter() - t0
    return scores, data.y.cpu().numpy().ravel()


@torch.no_grad()
def predict_embeddings(model, data, device: str = "cpu", chunk_nodes: Optional[int] = None) -> torch.Tensor:
    model.eval()
    if chunk_nodes is None or data.num_nodes <= chunk_nodes:
        return model.embeddings(data.to(device))
    out = []
    for idx in iter_node_chunks(data.num_nodes, chunk_nodes):
        out.append(model.embeddings(data.subgraph(idx).to(device)))
    return torch.cat(out)


def _ser(o):
    if isinstance(o, np.ndarray):
        return o.tolist()
    if isinstance(o, (np.floating,)):
        return float(o)
    if isinstance(o, (np.integer,)):
        return int(o)
    if isinstance(o, dict):
        return {k: _ser(v) for k, v in o.items() if not str(k).startswith("_")}
    if isinstance(o, (list, tuple)):
        return [_ser(v) for v in o]
    return o

# src/eval/metrics.py — AUC-ROC, AUC-PR, F1, Recall@k% (Paper V-A-3) + precision/FPR.
#   AUC-PR = average precision (sklearn).
#   Recall@k%: tỉ lệ fraud nằm trong top k% giao dịch có điểm nghi ngờ cao nhất.
# ============================================================
from __future__ import annotations

import numpy as np
from sklearn.metrics import average_precision_score, roc_auc_score


def recall_at_k_percent(y_true, y_score, k_percent: float = 1.0) -> float:
    y_true, y_score = np.asarray(y_true).astype(int), np.asarray(y_score, dtype=float)
    n, n_pos = len(y_true), int(y_true.sum())
    if n == 0 or n_pos == 0:
        return 0.0
    k = max(1, int(np.ceil(n * k_percent / 100.0)))
    top = np.argpartition(-y_score, min(k, n) - 1)[:k]
    return float(y_true[top].sum() / n_pos)


def _auc(y, s):
    if len(np.unique(y)) < 2:
        return 0.5, float(np.mean(y)) if len(y) else 0.0
    return float(roc_auc_score(y, s)), float(average_precision_score(y, s))


def classification_metrics(y_true, y_score, threshold: float = 0.5, y_pred=None) -> dict:
    """y_pred (tuỳ chọn) ghi đè quyết định (dùng khi threshold thay đổi theo batch)."""
    y = np.asarray(y_true).astype(int)
    s = np.asarray(y_score, dtype=float)
    ok = np.isfinite(s)
    if not ok.all():
        y, s = y[ok], s[ok]
        y_pred = None if y_pred is None else np.asarray(y_pred)[ok]
    if len(y) == 0:
        return {k: 0.0 for k in ("auc_roc", "auc_pr", "f1", "precision", "recall", "recall_at_1pct", "fpr")}
    pred = (s >= threshold) if y_pred is None else np.asarray(y_pred).astype(bool)
    tp = int(np.sum(pred & (y == 1))); fp = int(np.sum(pred & (y == 0)))
    fn = int(np.sum(~pred & (y == 1))); tn = int(np.sum(~pred & (y == 0)))
    prec, rec = tp / max(1, tp + fp), tp / max(1, tp + fn)
    auc_roc, auc_pr = _auc(y, s)
    return {
        "threshold": float(threshold), "auc_roc": auc_roc, "auc_pr": auc_pr,
        "f1": 2 * prec * rec / (prec + rec) if prec + rec > 0 else 0.0,
        "precision": prec, "recall": rec, "fpr": fp / max(1, fp + tn), "fnr": fn / max(1, fn + tp),
        "recall_at_1pct": recall_at_k_percent(y, s, 1.0),
        "tp": tp, "fp": fp, "tn": tn, "fn": fn,
        "num_samples": int(len(y)), "num_fraud": int(y.sum()),
    }


def best_f1_threshold(y_true, y_score, grid=None) -> float:
    """Ngưỡng tối đa F1 trên một tập (dùng cho baseline không có RL)."""
    from sklearn.metrics import precision_recall_curve
    p, r, t = precision_recall_curve(np.asarray(y_true).astype(int), np.asarray(y_score, dtype=float))
    f1 = 2 * p[:-1] * r[:-1] / np.clip(p[:-1] + r[:-1], 1e-12, None)
    return float(t[int(np.argmax(f1))]) if len(t) else 0.5

# ============================================================
# src/graph/graph_utils.py — Graph construction (Paper V-A-4):
#   "each transaction is a node ... we connect transactions that occur within
#    a 1-hour window and have a cosine similarity of their feature vectors
#    above a threshold of 0.9."
#
# Cài đặt TÌM KIẾM CHÍNH XÁC (không kNN xấp xỉ): sort theo thời gian, với mỗi
# chunk query chỉ so với các ứng viên trong [t-w, t+w] bằng phép nhân ma trận.
#   - Không có time (CC2023): chỉ dùng cosine, so với toàn bộ node.
#   - max_neighbors_per_node=None : đúng paper (không giới hạn).
#   - max_neighbors_per_node=k    : cap KỸ THUẬT (top-k theo cosine) để tránh nổ bộ nhớ
#                                   trên dataset rất lớn [DEVIATION — ghi rõ trong kết quả].
# Hướng cạnh: UNDIRECTED, không self-loop [INFERRED].
# ============================================================
from __future__ import annotations

from typing import Optional, Tuple

import numpy as np
import torch


def normalize_time_to_hours(time_values: Optional[np.ndarray], time_unit: Optional[str]) -> Optional[np.ndarray]:
    """Chuẩn hoá thời gian về GIỜ. time_unit: second | minute | hour | None (không có thời gian)."""
    if time_values is None or time_unit is None:
        return None
    t = np.asarray(time_values, dtype=np.float64)
    div = {"second": 3600.0, "minute": 60.0, "hour": 1.0}.get(time_unit)
    if div is None:
        raise ValueError(f"time_unit '{time_unit}' không hỗ trợ (second|minute|hour).")
    return (t / div).astype(np.float64)


@torch.no_grad()
def cosine_neighbors(
    x: np.ndarray | torch.Tensor,
    times_hours: Optional[np.ndarray] = None,
    threshold: float = 0.9,
    window_hours: Optional[float] = 1.0,
    max_neighbors: Optional[int] = None,
    causal: bool = False,
    device: str = "cpu",
    chunk_elems: float = 2e7,
    max_query_chunk: int = 4096,
    verbose: bool = True,
) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
    """
    Với mỗi node i, tìm các j != i thoả: cos(x_i,x_j) >= threshold và |t_i-t_j| <= window.

    Returns (edge_index [2,E], delta_hours [E], sim [E]); cạnh j -> i (src=j, dst=i).
    causal=True: chỉ giữ j đứng TRƯỚC i theo thời gian (dùng cho soft edges).
    """
    x = torch.as_tensor(np.asarray(x) if not torch.is_tensor(x) else x, dtype=torch.float32)
    n = x.size(0)
    xn = torch.nn.functional.normalize(x, dim=1, eps=1e-12).to(device)

    use_time = times_hours is not None and window_hours is not None
    if use_time:
        t_np = np.asarray(times_hours, dtype=np.float64)
        order_np = np.argsort(t_np, kind="stable")
        ts_np = t_np[order_np]
        lo_arr = np.searchsorted(ts_np, ts_np - window_hours, side="left")
        hi_arr = np.searchsorted(ts_np, ts_np + window_hours, side="right")
        order = torch.as_tensor(order_np, device=device)
        xs = xn[order]
        ts = torch.as_tensor(ts_np, dtype=torch.float64, device=device)
    else:
        order = torch.arange(n, device=device)
        xs, ts = xn, None
        lo_arr, hi_arr = np.zeros(n, dtype=np.int64), np.full(n, n, dtype=np.int64)

    src_l, dst_l, dlt_l, sim_l = [], [], [], []
    a, last_pct = 0, -1
    while a < n:
        q = min(max_query_chunk, n - a)
        while True:
            b = a + q
            lo, hi = int(lo_arr[a]), int(hi_arr[b - 1])  # lo,hi đơn điệu vì đã sort
            if q * (hi - lo) <= chunk_elems or q == 1:
                break
            q = max(1, q // 2)
        sims = xs[a:b] @ xs[lo:hi].T                                  # [q, c]
        mask = sims >= threshold
        r = torch.arange(q, device=device)
        self_col = (a + r) - lo                                       # vị trí chính node i trong ứng viên
        mask[r, self_col] = False
        if use_time:
            dt = (ts[a:b, None] - ts[None, lo:hi]).abs()
            mask &= dt <= window_hours
        if causal:  # ứng viên có chỉ số sort nhỏ hơn => không muộn hơn
            cand_idx = torch.arange(lo, hi, device=device)[None, :]
            mask &= cand_idx < (a + r)[:, None]
        if max_neighbors is not None and mask.any():
            k = min(int(max_neighbors), sims.size(1))
            vals, cols = sims.masked_fill(~mask, float("-inf")).topk(k, dim=1)
            rr, kk = (vals > float("-inf")).nonzero(as_tuple=True)
            cc = cols[rr, kk]
        else:
            rr, cc = mask.nonzero(as_tuple=True)
        if rr.numel():
            dst_l.append(order[a + rr].cpu())
            src_l.append(order[lo + cc].cpu())
            sim_l.append(sims[rr, cc].cpu())
            dlt_l.append((ts[a + rr] - ts[lo + cc]).abs().float().cpu() if use_time
                         else torch.zeros(rr.numel()))
        a = b
        if verbose:
            pct = int(100 * a / n)
            if pct // 20 > last_pct // 20:
                last_pct = pct
                print(f"[GRAPH]   {pct}% nodes scanned, edges so far={sum(len(s) for s in src_l):,}")

    if not src_l:
        e = torch.empty(0, dtype=torch.long)
        return torch.empty(2, 0, dtype=torch.long), torch.empty(0), torch.empty(0)
    return (torch.stack([torch.cat(src_l), torch.cat(dst_l)]).long(),
            torch.cat(dlt_l).float(), torch.cat(sim_l).float())


def dedupe_edges(edge_index: torch.Tensor, *attrs: torch.Tensor, num_nodes: int):
    """Loại cạnh trùng (giữ lần xuất hiện đầu) và các attr tương ứng."""
    if edge_index.size(1) == 0:
        return (edge_index, *attrs)
    key = edge_index[0] * num_nodes + edge_index[1]
    uniq, inv = torch.unique(key, return_inverse=True)
    first = torch.full((uniq.numel(),), edge_index.size(1), dtype=torch.long)
    first.scatter_reduce_(0, inv, torch.arange(edge_index.size(1)), reduce="amin")
    return (edge_index[:, first], *[a[first] for a in attrs])


def make_undirected(edge_index: torch.Tensor, delta: torch.Tensor, num_nodes: int):
    """Thêm cạnh ngược và loại trùng (cần khi dùng cap top-k vì quan hệ top-k không đối xứng)."""
    if edge_index.size(1) == 0:
        return edge_index, delta
    ei = torch.cat([edge_index, edge_index.flip(0)], dim=1)
    d = torch.cat([delta, delta])
    ei, d = dedupe_edges(ei, d, num_nodes=num_nodes)
    keep = ei[0] != ei[1]  # không self-loop
    return ei[:, keep], d[keep]

def _norm01(s_tr, s):
    lo, hi = float(np.min(s_tr)), float(np.max(s_tr))
    return np.clip((s - lo) / max(hi - lo, 1e-12), 0.0, None)

def run_xgboost(X, y, **kw):
    from xgboost import XGBClassifier
    m = XGBClassifier(n_estimators=300, max_depth=6, learning_rate=0.1, tree_method="hist",
                      scale_pos_weight=float((y[0] == 0).sum() / max(1, (y[0] == 1).sum())),
                      eval_metric="aucpr", random_state=kw.get("seed", 42), n_jobs=-1)

    tr = -m.score_samples(X[0][idx])

class _AE(nn.Module):
    def __init__(self, d):
        super().__init__()
        self.enc = nn.Sequential(nn.Linear(d, 64), nn.ReLU(), nn.Linear(64, 32), nn.ReLU(), nn.Linear(32, 16))
        self.dec = nn.Sequential(nn.Linear(16, 32), nn.ReLU(), nn.Linear(32, 64), nn.ReLU(), nn.Linear(64, d))

    def forward(self, x):
        return self.dec(self.enc(x))


def run_deepae(X, y, device="cpu", epochs=30, **kw):
    torch.manual_seed(kw.get("seed", 42))
    xt = torch.tensor(X[0][y[0] == 0], dtype=torch.float32)
    m = _AE(xt.size(1)).to(device)
    opt = torch.optim.Adam(m.parameters(), lr=1e-3)
    for _ in range(epochs):
        perm = torch.randperm(len(xt))
        for i in range(0, len(xt), 256):
            b = xt[perm[i:i + 256]].to(device)
            loss = F.mse_loss(m(b), b)
            opt.zero_grad(); loss.backward(); opt.step()
    m.eval()

    @torch.no_grad()
    def err(x):
        out = []
        for i in range(0, len(x), 4096):
            b = torch.tensor(x[i:i + 4096], dtype=torch.float32, device=device)
            out.append(((m(b) - b) ** 2).mean(1).cpu().numpy())
        return np.concatenate(out)

    tr = err(X[0])
    return [_norm01(tr, err(x)) for x in X[1:]]


class _GNN(nn.Module):
    def __init__(self, d, kind="gcn", hidden=64, layers=3, dropout=0.2, recon=False):
        super().__init__()
        conv = (lambda i, o: GCNConv(i, o)) if kind != "sage" else (lambda i, o: SAGEConv(i, o))
        self.recon = recon
        self.enc = nn.Linear(d, hidden) if recon else None
        self.dec = nn.Linear(hidden, d) if recon else None
        dims = [hidden if recon else d] + [hidden] * layers
        self.convs = nn.ModuleList([conv(dims[i], dims[i + 1]) for i in range(layers)])
        self.bns = nn.ModuleList([nn.BatchNorm1d(hidden) for _ in range(layers)])
        self.out = nn.Linear(hidden, 1)
        self.dp = dropout

    def forward(self, data):
        x = data.x
        rec = None
        if self.recon:
            z = F.relu(self.enc(x)); rec = self.dec(z); x = z
        for c, b in zip(self.convs, self.bns):
            x = F.dropout(F.relu(b(c(x, data.edge_index))), self.dp, self.training)
        return self.out(x).squeeze(-1), rec


def run_graph_baseline(graphs, kind: str, device="cpu", epochs=100, patience=10, lr=1e-3, recon_lambda=0.1, **kw):
    from src.eval.evaluate import predict_scores  # noqa
    torch.manual_seed(kw.get("seed", 42))
    g_tr, g_va, g_te = [g.to(device) for g in graphs]
    model = _GNN(g_tr.x.size(1), "sage" if kind == "graphsemi" else "gcn", recon=(kind == "autogcn")).to(device)
    opt = torch.optim.Adam(model.parameters(), lr=lr)
    pw = compute_pos_weight(g_tr.y, kw.get("pos_weight_max", 100.0)).to(device)

    @torch.no_grad()
    def score(g):
        model.eval()
        return torch.sigmoid(model(g)[0]).cpu().numpy()

    best, best_state, bad = -1.0, None, 0
    for ep in range(epochs):
        model.train()
        opt.zero_grad()
        logit, rec = model(g_tr)
        loss = F.binary_cross_entropy_with_logits(logit, g_tr.y.float(), pos_weight=pw)
        if rec is not None:
            loss = loss + recon_lambda * F.mse_loss(rec, g_tr.x)
        loss.backward(); opt.step()
        m = classification_metrics(g_va.y.cpu().numpy(), score(g_va), 0.5)
        if m["auc_pr"] > best:
            best, bad, best_state = m["auc_pr"], 0, copy.deepcopy(model.state_dict())
        else:
            bad += 1
            if bad >= patience:
                break
    model.load_state_dict(best_state)
    return [score(g_va), score(g_te)]


class _CNN(nn.Module):
    def __init__(self, d):
        super().__init__()
        self.conv = nn.Conv1d(1, 16, 3, padding=1)
        self.fc = nn.Sequential(nn.Linear(16 * (d // 2), 64), nn.ReLU(), nn.Linear(64, 1))

    def forward(self, x):
        h = F.max_pool1d(F.relu(self.conv(x.unsqueeze(1))), 2)
        return self.fc(h.flatten(1)).squeeze(-1)


def run_fedfraud(X, y, device="cpu", rounds=30, num_clients=3, lr=1e-3, **kw):

    pad = lambda a: np.pad(a, ((0, 0), (0, d_eff - d))) if d_eff != d else a
    Xs = [torch.tensor(pad(a), dtype=torch.float32) for a in X]
    ys = torch.tensor(y[0], dtype=torch.float32)
    shards = np.array_split(np.arange(len(Xs[0])), num_clients)       # shard theo thứ tự (đã sort thời gian)
    glob = _CNN(d_eff).to(device)
    worker = copy.deepcopy(glob)
    pw = compute_pos_weight(ys.long(), 100.0).to(device)
    state = copy.deepcopy(glob.state_dict())
    for _ in range(rounds):
        sts = []
        for sh in shards:
            worker.load_state_dict(state); worker.train()
            opt = torch.optim.Adam(worker.parameters(), lr=lr)
            for i in range(0, len(sh), 512):
                b = sh[i:i + 512]
                loss = F.binary_cross_entropy_with_logits(worker(Xs[0][b].to(device)), ys[b].to(device), pos_weight=pw)
                opt.zero_grad(); loss.backward(); opt.step()
            sts.append(copy.deepcopy(worker.state_dict()))
        state = fedavg_uniform(sts)
    glob.load_state_dict(state); glob.eval()

    @torch.no_grad()
    def sc(a):
        return np.concatenate([torch.sigmoid(glob(a[i:i + 4096].to(device))).cpu().numpy() for i in range(0, len(a), 4096)])

    return [sc(Xs[1]), sc(Xs[2])]

# ============================================================
# src/models/dqn_agent.py — Vanilla DQN (Eq. 3-4, 12) trên không gian hành động ĐÃ RỜI RẠC HOÁ.
# Dùng khi rl.type = "dqn": action = chỉ số bin của threshold; feature weights giữ đều (w=1).
# Mặc định của paper là NAF (naf_agent.py) vì "continuous action space".
# ============================================================
from __future__ import annotations

import random
from collections import deque
from typing import Tuple

import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F


class DQNAgent:
    def __init__(self, state_dim: int, n_features: int, n_bins: int = 19, hidden_dim: int = 128,
                 gamma: float = 0.99, lr: float = 1e-3, tau: float = 0.005, buffer_size: int = 10000,
                 batch_size: int = 64, eps_start: float = 1.0, eps_end: float = 0.05, eps_decay: float = 0.995,
                 device: str = "cpu"):
        self.kind = "dqn"
        self.device, self.n_features = device, n_features
        self.bins = np.linspace(0.05, 0.95, n_bins).astype(np.float32)
        mk = lambda: nn.Sequential(nn.Linear(state_dim, hidden_dim), nn.ReLU(),
                                   nn.Linear(hidden_dim, hidden_dim), nn.ReLU(),
                                   nn.Linear(hidden_dim, n_bins)).to(device)
        self.policy, self.target = mk(), mk()
        self.target.load_state_dict(self.policy.state_dict())
        self.opt = torch.optim.Adam(self.policy.parameters(), lr=lr)
        self.memory = deque(maxlen=buffer_size)
        self.gamma, self.tau, self.bs = gamma, tau, batch_size
        self.eps, self.eps_end, self.eps_decay = eps_start, eps_end, eps_decay
        self.n_bins = n_bins

    @torch.no_grad()
    def act(self, state: np.ndarray, explore: bool = True) -> Tuple[np.ndarray, float, np.ndarray]:
        if explore and random.random() < self.eps:
            k = random.randrange(self.n_bins)
        else:
            s = torch.as_tensor(state, dtype=torch.float32, device=self.device).unsqueeze(0)
            k = int(self.policy(s).argmax(1).item())
        fw = np.full(self.n_features, 1.0 / self.n_features, dtype=np.float32)
        return np.array([k], dtype=np.float32), float(self.bins[k]), fw

    def remember(self, s, a, r, s2, d):
        self.memory.append((s, a, r, s2, d))

    def update(self):
        if len(self.memory) < self.bs:
            return None
        s, a, r, s2, d = zip(*random.sample(self.memory, self.bs))
        f = lambda z, dt=torch.float32: torch.as_tensor(np.asarray(z), dtype=dt, device=self.device)
        s, s2, r, d = f(s), f(s2), f(r), f(d)
        a = f(a, torch.long).view(-1)
        q = self.policy(s).gather(1, a[:, None]).squeeze(1)
        with torch.no_grad():
            y = r + self.gamma * self.target(s2).max(1).values * (1.0 - d)   # Eq. 12
        loss = F.mse_loss(q, y)
        self.opt.zero_grad(); loss.backward()
        nn.utils.clip_grad_norm_(self.policy.parameters(), 5.0)
        self.opt.step()
        with torch.no_grad():
            for tp, pp in zip(self.target.parameters(), self.policy.parameters()):
                tp.mul_(1 - self.tau).add_(self.tau * pp)
        self.eps = max(self.eps_end, self.eps * self.eps_decay)
        return float(loss.item())


__all__ = ["DQNAgent"]

# ============================================================
# src/models/fraudgnn_rl.py — encoder (TSSGC | MLP) + classifier + feature importance weights.
#
# Feature importance weights (Action của RL, IV-B) nhân vào embedding TRƯỚC classifier.
# Dạng chuẩn hoá: w = D * softmax-like(action)  -> trung bình = 1 (trung tính khi đều),
# nên classifier huấn luyện với w = 1 vẫn dùng được khi RL đổi trọng số.
#
# shared_projection (Graph alignment cho FL, IV-C) [INFERRED, đã xác nhận phương án B]:
#   Linear(in_dim -> hidden) dùng chung, được FedAvg. Áp dụng ở MỌI chế độ (FL & non-FL)
#   để ablation 'w/o FL' chỉ khác ở cách train.
# ============================================================
from __future__ import annotations

import torch
import torch.nn as nn

from .classifier import FraudClassifier
from .tssgc import MLPEncoder, TSSGCEncoder


class FraudGNNRL(nn.Module):
    def __init__(self, in_dim: int, hidden_dim: int = 64, num_layers: int = 3, num_node_types: int = 1,
                 dropout: float = 0.2, use_gnn: bool = True, shared_projection: bool = True,
                 temporal_max_seq_len: int | None = None):
        super().__init__()
        self.hidden_dim, self.use_gnn = hidden_dim, use_gnn
        self.projection = nn.Linear(in_dim, hidden_dim) if shared_projection else nn.Identity()
        enc_in = hidden_dim if shared_projection else in_dim
        self.encoder = (TSSGCEncoder(enc_in, hidden_dim, num_layers, num_node_types, dropout, temporal_max_seq_len)
                        if use_gnn else MLPEncoder(enc_in, hidden_dim, num_layers, dropout))
        self.classifier = FraudClassifier(hidden_dim, hidden_dim, dropout)
        self.register_buffer("feature_weights", torch.ones(hidden_dim))  # cập nhật bởi RL, không qua gradient

    # -- RL hooks -----------------------------------------------------
    def set_feature_weights(self, w: torch.Tensor) -> None:
        w = torch.as_tensor(w, dtype=torch.float32, device=self.feature_weights.device).flatten()
        if w.numel() != self.hidden_dim:
            raise ValueError(f"feature weights cần {self.hidden_dim} phần tử, nhận {w.numel()}")
        self.feature_weights.copy_(w)

    def reset_feature_weights(self) -> None:
        self.feature_weights.fill_(1.0)

    # -- forward ------------------------------------------------------
    def encode(self, data) -> torch.Tensor:
        dev = self.feature_weights.device
        x = self.projection(data.x.to(dev))
        ei = data.edge_index.to(dev)
        dt = getattr(data, "edge_time_delta", None)
        nt = getattr(data, "node_type", None)
        return self.encoder(x, ei, None if dt is None else dt.to(dev), None if nt is None else nt.to(dev))

    def logits_from_embedding(self, emb: torch.Tensor, weights: torch.Tensor | None = None) -> torch.Tensor:
        w = self.feature_weights if weights is None else weights
        return self.classifier(emb * w)

    def forward(self, data) -> torch.Tensor:
        return self.logits_from_embedding(self.encode(data))

    @torch.no_grad()
    def embeddings(self, data) -> torch.Tensor:
        """Embedding TRƯỚC khi nhân feature weights (State của RL)."""
        return self.encode(data)


__all__ = ["FraudGNNRL"]

# ============================================================
# src/models/naf_agent.py — Normalized Advantage Functions (Gu et al., 2016) [29].
#
# Paper IV-B: "The action space is continuous, representing the fraud detection threshold
# and feature importance weights. We discretize this space and use a variant of DQN called
# Deep Q-Learning with Normalized Advantage Functions (NAF) to handle the continuous action
# space more effectively."   Loss = Eq. 12 (target network theta^-).
#
#   Q(s,a) = V(s) + A(s,a),  A = -1/2 (a-mu(s))^T P(s) (a-mu(s)),  P = L L^T
#   Action a in [0,1]^(1+D): a[0]=threshold, a[1:]=raw feature weights (mu qua sigmoid => cùng không gian).
#
# Siêu tham số paper nêu: 2 hidden layers x 128, Adam lr=1e-3.
# [INFERRED]: gamma=0.99, replay 10k, Polyak tau=0.005, Gaussian noise decay 0.995, update-batch 64.
# ============================================================
from __future__ import annotations

import random
from collections import deque
from typing import Tuple

import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F


class ReplayBuffer:
    def __init__(self, capacity: int = 10000):
        self.buf = deque(maxlen=capacity)

    def push(self, s, a, r, s2, d):
        self.buf.append((s, a, r, s2, d))

    def sample(self, n: int, device):
        batch = random.sample(self.buf, n)
        s, a, r, s2, d = zip(*batch)
        f = lambda z: torch.as_tensor(np.asarray(z), dtype=torch.float32, device=device)
        return f(s), f(a), f(r), f(s2), f(d)

    def __len__(self):
        return len(self.buf)


class NAFNetwork(nn.Module):
    def __init__(self, state_dim: int, action_dim: int, hidden: int = 128):
        super().__init__()
        self.A = action_dim
        self.trunk = nn.Sequential(nn.Linear(state_dim, hidden), nn.ReLU(), nn.Linear(hidden, hidden), nn.ReLU())
        self.v = nn.Linear(hidden, 1)
        self.mu = nn.Linear(hidden, action_dim)
        self.l_diag = nn.Linear(hidden, action_dim)
        self.n_off = action_dim * (action_dim - 1) // 2
        self.l_off = nn.Linear(hidden, self.n_off)
        self.register_buffer("tril", torch.tril_indices(action_dim, action_dim, offset=-1))

    def forward(self, s: torch.Tensor, a: torch.Tensor | None = None):
        h = self.trunk(s)
        v, mu = self.v(h).squeeze(-1), torch.sigmoid(self.mu(h))
        if a is None:
            return v, mu, None
        B = s.size(0)
        L = s.new_zeros(B, self.A, self.A)
        idx = torch.arange(self.A, device=s.device)
        L[:, idx, idx] = torch.exp(self.l_diag(h).clamp(-4.0, 2.0))
        if self.n_off:
            L[:, self.tril[0], self.tril[1]] = self.l_off(h)
        y = torch.bmm(L.transpose(1, 2), (a - mu).unsqueeze(-1)).squeeze(-1)   # L^T (a-mu)
        adv = -0.5 * (y ** 2).sum(-1)                                           # = -1/2 d^T P d
        return v, mu, v + adv


class NAFAgent:
    def __init__(self, state_dim: int, n_features: int, hidden_dim: int = 128, gamma: float = 0.99,
                 lr: float = 1e-3, tau: float = 0.005, buffer_size: int = 10000, batch_size: int = 64,
                 noise_std: float = 0.10, noise_decay: float = 0.995, noise_min: float = 0.01,
                 device: str = "cpu"):
        self.kind = "naf"
        self.device = device
        self.n_features = n_features
        self.A = 1 + n_features
        self.gamma, self.tau, self.bs = gamma, tau, batch_size
        self.policy = NAFNetwork(state_dim, self.A, hidden_dim).to(device)
        self.target = NAFNetwork(state_dim, self.A, hidden_dim).to(device)
        self.target.load_state_dict(self.policy.state_dict())
        self.opt = torch.optim.Adam(self.policy.parameters(), lr=lr)
        self.memory = ReplayBuffer(buffer_size)
        self.noise_std, self.noise_decay, self.noise_min = noise_std, noise_decay, noise_min

    def remember(self, s, a, r, s2, d):
        self.memory.push(s, a, r, s2, d)

    @torch.no_grad()
    def act(self, state: np.ndarray, explore: bool = True) -> Tuple[np.ndarray, float, np.ndarray]:
        """Returns (action_vector, threshold, fw_raw)."""
        s = torch.as_tensor(state, dtype=torch.float32, device=self.device).unsqueeze(0)
        _, mu, _ = self.policy(s)
        a = mu[0].cpu().numpy()
        if explore:
            a = np.clip(a + np.random.normal(0, self.noise_std, size=a.shape), 0.0, 1.0)
        a = a.astype(np.float32)
        return a, float(a[0]), a[1:]

    def update(self):
        if len(self.memory) < self.bs:
            return None
        s, a, r, s2, d = self.memory.sample(self.bs, self.device)
        _, _, q = self.policy(s, a)
        with torch.no_grad():
            v2, _, _ = self.target(s2)
            y = r + self.gamma * v2 * (1.0 - d)               # Eq. 12 với V(s') = max_a' Q(s',a')
        loss = F.mse_loss(q, y)
        self.opt.zero_grad()
        loss.backward()
        nn.utils.clip_grad_norm_(self.policy.parameters(), 5.0)
        self.opt.step()
        with torch.no_grad():
            for tp, pp in zip(self.target.parameters(), self.policy.parameters()):
                tp.mul_(1 - self.tau).add_(self.tau * pp)
        self.noise_std = max(self.noise_min, self.noise_std * self.noise_decay)
        return float(loss.item())


__all__ = ["ReplayBuffer", "NAFNetwork", "NAFAgent"]

# ============================================================
# src/models/reward.py — Reward + môi trường RL (Section IV-B / IV-D).
#
# Paper: State = graph embedding (TSSGC); Action = (fraud threshold, feature importance
# weights); Reward = "combination of detection accuracy and false positive rate".
# Paper KHÔNG cho công thức -> [INFERRED]:
#       reward = F1 + recall_weight * Recall - fpr_penalty * FPR
#   (accuracy thuần bị chi phối bởi lớp normal khi fraud ~0.1% nên dùng F1/Recall làm "detection accuracy")
#
# Môi trường (Fig. 2: Update Graph -> TSSGC -> RL action -> Classification -> Feedback -> Update policy):
#   - 1 episode = quét tuần tự các batch KHÔNG chồng lấn trên tập (theo thứ tự thời gian).
#   - state  = mean pooling embedding (trước feature weights) của batch hiện tại.
#   - action = [threshold, raw feature weights] ; w = D * a_fw / sum(a_fw).
#   - score  = sigmoid(classifier(embedding * w)) => feature weights THỰC SỰ ảnh hưởng điểm số/reward.
# ============================================================
from __future__ import annotations

from typing import Dict, Optional, Tuple

import numpy as np
import torch


def compute_reward(tp: int, fp: int, tn: int, fn: int, fpr_penalty: float = 1.0,
                   recall_weight: float = 0.25) -> float:
    precision = tp / max(1, tp + fp)
    recall = tp / max(1, tp + fn)
    f1 = 2 * precision * recall / (precision + recall) if precision + recall > 0 else 0.0
    fpr = fp / max(1, fp + tn)
    return float(f1 + recall_weight * recall - fpr_penalty * fpr)


def weights_from_action(fw_raw: np.ndarray) -> np.ndarray:
    """a_fw in [0,1]^D  ->  w, mean(w)=1."""
    fw = np.clip(np.asarray(fw_raw, dtype=np.float64), 1e-6, None)
    return (fw.size * fw / fw.sum()).astype(np.float32)


class BatchFraudEnv:
    """
    embeddings: [N, D] (tensor, đã ở thiết bị tính toán) ; labels: [N] ;
    classifier: FraudClassifier (eval, đóng băng) dùng để chấm điểm batch.
    """

    def __init__(self, embeddings: torch.Tensor, labels: np.ndarray, classifier: torch.nn.Module,
                 batch_size: int = 256, fpr_penalty: float = 1.0, recall_weight: float = 0.25):
        self.emb = embeddings
        self.labels = np.asarray(labels, dtype=np.int64)
        self.clf = classifier.eval()
        self.bs, self.fpr_penalty, self.recall_weight = int(batch_size), float(fpr_penalty), float(recall_weight)
        self.n, self.state_dim = embeddings.size(0), embeddings.size(1)
        self.pos = 0

    def _state(self) -> np.ndarray:
        if self.pos >= self.n:
            return np.zeros(self.state_dim, dtype=np.float32)
        return self.emb[self.pos:self.pos + self.bs].mean(0).float().cpu().numpy()

    def reset(self) -> np.ndarray:
        self.pos = 0
        return self._state()

    @torch.no_grad()
    def step(self, threshold: float, weights: Optional[np.ndarray] = None
             ) -> Tuple[np.ndarray, float, bool, Dict]:
        a, b = self.pos, min(self.n, self.pos + self.bs)
        e = self.emb[a:b]
        if weights is not None:
            e = e * torch.as_tensor(weights, dtype=e.dtype, device=e.device)
        s = torch.sigmoid(self.clf(e)).cpu().numpy()
        y = self.labels[a:b]
        pred = s >= threshold
        tp = int(np.sum(pred & (y == 1))); fp = int(np.sum(pred & (y == 0)))
        fn = int(np.sum(~pred & (y == 1))); tn = int(np.sum(~pred & (y == 0)))
        r = compute_reward(tp, fp, tn, fn, self.fpr_penalty, self.recall_weight)
        self.pos = b
        done = self.pos >= self.n
        info = dict(tp=tp, fp=fp, tn=tn, fn=fn, threshold=float(threshold), start=a, end=b)
        return self._state(), r, done, info


__all__ = ["compute_reward", "weights_from_action", "BatchFraudEnv"]

# ============================================================
# src/models/tssgc.py — TSSGC (Paper IV-A, Eq. 5-11)
#
#   Eq. 5/11 : h_i^(l+1) = sigma( W_t.TEMP(i) + W_s.SPAT(i) + W_m.SEM(i) + b )
#   Eq. 6    : TEMP(i)  = GRU({(f_k, alpha_k) | (v_i,v_j,t_k,f_k) in E_i})
#   Eq. 7    : alpha_k  = exp(-beta (t_now - t_k)) / sum_l exp(-beta (t_now - t_l))
#   Eq. 8-9  : SPAT(i)  = sum_j alpha_ij W_s h_j,  alpha_ij = softmax(LeakyReLU(a^T[W_s h_i || W_s h_j]))
#   Eq. 10   : SEM(i)   = W_m [h_i || e_type(i)]
#   V-A-4    : 3 lớp, 64 hidden, ReLU, BatchNorm sau mỗi lớp conv.
#
# Lựa chọn [INFERRED] (paper không nói):
#   - beta = softplus(beta_raw) > 0 (learnable, Eq. 7 nói "learnable parameter").
#   - Dãy cạnh đưa vào GRU: sort theo |dt| giảm dần (cũ -> mới), đầu vào = alpha_k * W_msg(x_j);
#     lấy hidden bước cuối.
#   - |dt| = |t_i - t_j| của cạnh (graph vô hướng).
#   - SPAT dùng GATConv 1 head (bám Eq. 8-9 thay vì chữ "GCN" ở văn xuôi), không self-loop.
#   - BatchNorm đặt sau khi cộng 3 nhánh + bias, trước ReLU.
#   - temporal_max_seq_len (None = paper): cap kỹ thuật, giữ N cạnh gần nhất cho GRU.
# ============================================================
from __future__ import annotations

import warnings
from typing import Optional

import torch
import torch.nn as nn
import torch.nn.functional as F
from torch_geometric.nn import GATConv
from torch_geometric.utils import softmax as pyg_softmax


class TemporalAggregator(nn.Module):
    def __init__(self, in_dim: int, out_dim: int, max_seq_len: Optional[int] = None):
        super().__init__()
        self.out_dim = out_dim
        self.max_seq_len = max_seq_len
        self.lin_msg = nn.Linear(in_dim, out_dim)
        self.gru = nn.GRU(out_dim, out_dim, batch_first=True)
        self.beta_raw = nn.Parameter(torch.tensor(0.0))

    def forward(self, x, edge_index, edge_time_delta):
        N, D, dev = x.size(0), self.out_dim, x.device
        E = edge_index.size(1)
        if E == 0:
            return x.new_zeros(N, D)
        src, dst = edge_index
        msg = self.lin_msg(x[src])
        delta = (edge_time_delta.to(dev).to(msg.dtype).clamp(min=0)
                 if edge_time_delta is not None and edge_time_delta.numel() == E
                 else torch.zeros(E, device=dev, dtype=msg.dtype))

        alpha = pyg_softmax(-F.softplus(self.beta_raw) * delta, dst, num_nodes=N)  # Eq. 7
        msg = msg * alpha.unsqueeze(-1)

        # sắp xếp: theo dst, trong mỗi dst từ CŨ (delta lớn) -> MỚI (delta nhỏ)
        o1 = torch.argsort(delta, descending=True, stable=True)
        order = o1[torch.argsort(dst[o1], stable=True)]
        dst_s, msg_s = dst[order], msg[order]

        deg = torch.bincount(dst_s, minlength=N)
        ptr = torch.cumsum(deg, 0) - deg
        pos = torch.arange(E, device=dev) - ptr[dst_s]
        if self.max_seq_len is not None:  # giữ các cạnh MỚI nhất
            drop = (deg - self.max_seq_len).clamp(min=0)
            keep = pos >= drop[dst_s]
            dst_s, msg_s, pos = dst_s[keep], msg_s[keep], pos[keep] - drop[dst_s][keep]
            deg = deg.clamp(max=self.max_seq_len)

        nodes = (deg > 0).nonzero(as_tuple=True)[0]
        row = torch.full((N,), -1, dtype=torch.long, device=dev)
        row[nodes] = torch.arange(nodes.numel(), device=dev)
        L = int(deg.max().item())
        padded = msg_s.new_zeros(nodes.numel(), L, D)
        padded[row[dst_s], pos] = msg_s
        out, _ = self.gru(padded)                                   # GRU nhân quả: đệm 0 phía sau không ảnh hưởng
        last = out[torch.arange(nodes.numel(), device=dev), deg[nodes] - 1]
        h = x.new_zeros(N, D)
        h[nodes] = last
        return h


class SemanticEncoder(nn.Module):
    """Eq. 10: SEM(i) = W_m [h_i || e_type(i)]"""

    def __init__(self, in_dim: int, out_dim: int, num_node_types: int = 1):
        super().__init__()
        self.num_node_types = max(1, num_node_types)
        self.type_emb = nn.Embedding(self.num_node_types, out_dim)
        self.lin = nn.Linear(in_dim + out_dim, out_dim)

    def forward(self, x, node_type):
        if node_type is None:
            node_type = torch.zeros(x.size(0), dtype=torch.long, device=x.device)
        node_type = node_type.to(x.device).long()
        bad = int(((node_type < 0) | (node_type >= self.num_node_types)).sum())
        if bad:
            warnings.warn(f"SemanticEncoder: {bad} node_type ngoài phạm vi -> clamp.")
            node_type = node_type.clamp(0, self.num_node_types - 1)
        return self.lin(torch.cat([x, self.type_emb(node_type)], dim=-1))


class TSSGCLayer(nn.Module):
    def __init__(self, in_dim, out_dim, num_node_types=1, dropout=0.0, max_seq_len=None):
        super().__init__()
        self.temporal = TemporalAggregator(in_dim, out_dim, max_seq_len)
        self.spatial = GATConv(in_dim, out_dim, heads=1, concat=False, dropout=dropout, add_self_loops=False)
        self.semantic = SemanticEncoder(in_dim, out_dim, num_node_types)
        self.W_t = nn.Linear(out_dim, out_dim, bias=False)
        self.W_s = nn.Linear(out_dim, out_dim, bias=False)
        self.W_m = nn.Linear(out_dim, out_dim, bias=False)
        self.b = nn.Parameter(torch.zeros(out_dim))
        self.norm = nn.BatchNorm1d(out_dim)
        self.dropout = nn.Dropout(dropout)

    def forward(self, x, edge_index, edge_time_delta=None, node_type=None):
        out = (self.W_t(self.temporal(x, edge_index, edge_time_delta))
               + self.W_s(self.spatial(x, edge_index))
               + self.W_m(self.semantic(x, node_type)) + self.b)
        return self.dropout(F.relu(self.norm(out)))


class TSSGCEncoder(nn.Module):
    """3 lớp TSSGC, hidden 64 (paper V-A-4)."""

    def __init__(self, in_dim, hidden_dim=64, num_layers=3, num_node_types=1, dropout=0.0, max_seq_len=None):
        super().__init__()
        self.layers = nn.ModuleList([
            TSSGCLayer(in_dim if i == 0 else hidden_dim, hidden_dim, num_node_types, dropout, max_seq_len)
            for i in range(num_layers)
        ])

    def forward(self, x, edge_index, edge_time_delta=None, node_type=None):
        for layer in self.layers:
            x = layer(x, edge_index, edge_time_delta, node_type)
        return x


class MLPEncoder(nn.Module):
    """Encoder "standard deep neural network" cho ablation 'w/o GNN' (cùng độ sâu/độ rộng, không dùng cạnh)."""

    def __init__(self, in_dim, hidden_dim=64, num_layers=3, dropout=0.0):
        super().__init__()
        blocks = []
        for i in range(num_layers):
            blocks += [nn.Linear(in_dim if i == 0 else hidden_dim, hidden_dim),
                       nn.BatchNorm1d(hidden_dim), nn.ReLU(), nn.Dropout(dropout)]
        self.net = nn.Sequential(*blocks)

    def forward(self, x, edge_index=None, edge_time_delta=None, node_type=None):
        return self.net(x)

# src/train/federated.py — Federated Learning (Paper IV-C, Algorithm 1).
#
#   1: Initialize θ0
#   2: for t = 1..T:
#   3:    Select a subset of clients C_t
#   4:    for each client i in C_t in parallel:
#   5:        θ_t^i <- LocalUpdate(θ_{t-1}, G_i)        (G_i = local transaction graph)
#   7:    θ_t <- (1/|C_t|) Σ θ_t^i                      <- trung bình ĐỀU, không trọng số n_i
#
# Paper rõ:  FedAvg, trung bình đều, mỗi client có đồ thị cục bộ G_i, LocalUpdate = vài bước SGD,
#            "graph alignment technique" để nhất quán giữa các graph cục bộ.
# [INFERRED] (paper không nói):
#   - num_clients=3; chọn TẤT CẢ client mỗi round (|C_t|=K, tuỳ chọn client_fraction<1).
#   - Chia client theo THỜI GIAN (shard liên tục) -> mỗi client là một "tổ chức" có đồ thị riêng;
#     cạnh xuyên shard bị loại (không chia sẻ dữ liệu thô).
#   - Graph alignment = schema đặc trưng chung (cùng preprocessor) + projection layer dùng chung
#     được FedAvg (model.shared_projection) — phương án B đã xác nhận.
#   - rounds=50 x local_epochs=2 (= 100 epoch-tương-đương như paper), early stopping patience 10 round
#     trên AUC-PR của global model ở val.
#   - Optimizer Adam cục bộ giữ trạng thái qua các round; đồng bộ cả buffer (BatchNorm running stats).
# ============================================================
from __future__ import annotations

import numpy as np

def fedavg_uniform(states: List[OrderedDict]) -> OrderedDict:
    """θ = (1/|C_t|) Σ θ_i  (Algorithm 1, dòng 7)."""
    n = len(states)
    out = OrderedDict()
    for k in states[0]:
        stacked = torch.stack([s[k].float() for s in states])
        out[k] = stacked.mean(0).to(states[0][k].dtype) if states[0][k].is_floating_point() else stacked.mean(0).round().to(states[0][k].dtype)
    return out


class FederatedClient:
    def __init__(self, cid: int, data, cfg: Dict[str, Any], device: str):
        self.cid, self.data, self.cfg, self.device = cid, data.to(device), cfg, device
        self.pos_w = compute_pos_weight(data.y, cfg.get("train", {}).get("pos_weight_max", 100.0)).to(device)
        self.opt = None
        print(f"[FL] Client {cid}: N={data.num_nodes:,} edges={data.edge_index.size(1):,} "
              f"fraud={float((data.y==1).float().mean())*100:.4f}%")

    def local_update(self, global_state: OrderedDict, model, epochs: int, lr: float) -> OrderedDict:
        model.load_state_dict(global_state)
        if self.opt is None:
            self.opt = torch.optim.Adam(model.parameters(), lr=lr)
        else:  # giữ trạng thái Adam, trỏ lại vào tham số của model dùng chung
            self.opt.param_groups[0]["params"] = list(model.parameters())
        train_steps(model, self.data, self.opt, self.pos_w, self.cfg.get("train", {}), self.device, epochs=epochs)
        return copy.deepcopy(model.state_dict())


def create_federated_clients(train_data, cfg: Dict[str, Any], num_clients: int, device: str) -> List[FederatedClient]:
    n = train_data.num_nodes
    num_clients = max(1, min(num_clients, n))
    order = torch.argsort(train_data.node_time) if getattr(train_data, "has_time", False) else torch.arange(n)
    shard = n // num_clients
    clients = []
    for cid in range(num_clients):
        a, b = cid * shard, (cid + 1) * shard if cid < num_clients - 1 else n
        clients.append(FederatedClient(cid, train_data.subgraph(order[a:b].sort().values), cfg, device))
    return clients


def train_federated(train_data, val_data, cfg: Dict[str, Any], device: str, timing: dict | None = None) -> Dict[str, Any]:
    fl, t = cfg.get("federated", {}), cfg.get("train", {})
    flags = get_flags(cfg)
    rounds, local_epochs = int(fl.get("rounds", 50)), int(fl.get("local_epochs", 2))
    lr, patience = float(fl.get("learning_rate", t.get("learning_rate", 1e-3))), int(fl.get("patience", t.get("patience", 10)))
    frac = float(fl.get("client_fraction", 1.0))
    eval_chunk = t.get("eval_chunk_nodes")

    global_model = build_model(int(train_data.x.size(1)), cfg, device)
    worker = copy.deepcopy(global_model)               # model dùng chung cho LocalUpdate (tuần tự)
    clients = create_federated_clients(train_data, cfg, int(fl.get("num_clients", 3)), device)

    best_state, best, bad, hist, round_times = None, -1.0, 0, [], []
    g_state = copy.deepcopy(global_model.state_dict())
    t_start = time.perf_counter()
    for r in range(1, rounds + 1):
        rt0 = time.perf_counter()
        k = max(1, int(round(frac * len(clients))))
        chosen = clients if k == len(clients) else [clients[i] for i in np.random.choice(len(clients), k, replace=False)]
        states = [c.local_update(g_state, worker, local_epochs, lr) for c in chosen]
        g_state = fedavg_uniform(states)
        global_model.load_state_dict(g_state)
        g_state = copy.deepcopy(global_model.state_dict())
        s, y = predict_scores(global_model, val_data, device, chunk_nodes=eval_chunk)
        m = classification_metrics(y, s, 0.5)
        round_times.append(time.perf_counter() - rt0)
        hist.append({"round": r, "val_auc_pr": m["auc_pr"], "val_auc_roc": m["auc_roc"], "round_time_sec": round_times[-1]})
        if r == 1 or r % 5 == 0:
            print(f"[FL] round {r:3d}/{rounds} val AUC-PR={m['auc_pr']:.4f} AUC-ROC={m['auc_roc']:.4f} ({round_times[-1]:.1f}s)")
        if m["auc_pr"] > best:
            best, bad, best_state = m["auc_pr"], 0, copy.deepcopy(g_state)
        else:
            bad += 1
            if bad >= patience:
                print(f"[FL] Early stop @ round {r} (best val AUC-PR={best:.4f})")
                break
    if best_state is not None:
        global_model.load_state_dict(best_state)
    total = time.perf_counter() - t_start
    if timing is not None:
        timing["gnn_training_sec"] = total
    return {"model": global_model, "history": hist, "best_val_auc_pr": best, "total_time_sec": total,
            "avg_round_time_sec": float(np.mean(round_times)) if round_times else 0.0,
            "num_clients": len(clients), "num_rounds": len(hist)}

def run_pipeline_on_split(train_df, val_df, test_df, cfg: Dict[str, Any], prepared: Optional[dict] = None,
                          seed: Optional[int] = None, return_artifacts: bool = False):
    seed = int(seed if seed is not None else cfg.get("seed", cfg.get("dataset", {}).get("random_state", 42)))
    set_seed(seed)
    flags, device = get_flags(cfg), get_device(cfg)
    # Đã thay thế resolve_model_name(flags) bằng chuỗi cố định
    print("=" * 60 + f"\n[PIPE] FraudGNN-RL | flags={flags} | device={device} | seed={seed}\n" + "=" * 60)
    t_all = time.perf_counter()
    prep = prepared or prepare_splits(train_df, val_df, test_df, cfg)
    g_tr, g_va, g_te = prep["graphs"]
    timing = {"preprocessing_sec": 0.0, "graph_building_sec": 0.0, "gnn_training_sec": 0.0,
              "rl_training_sec": 0.0, "inference_sec": 0.0, **prep["timing"]}
    eval_chunk = cfg.get("train", {}).get("eval_chunk_nodes")

    # ---- 1) TSSGC + classifier (FL | centralized) ----------------------
    if flags["federated"]:
        fed = train_federated(g_tr, g_va, cfg, device, timing)
        model, hist = fed["model"], fed["history"]
        fl_info = {k: fed[k] for k in ("num_clients", "num_rounds", "avg_round_time_sec")}
    else:
        out = train_tssgc_classifier(g_tr, g_va, cfg, device, timing)
        model, hist = out["model"], out["history"]
        fl_info = {"centralized": True}
    model.reset_feature_weights()

    # ---- 2) RL ---------------------------------------------------------
    rl_cfg, threshold, rl_info, y_pred_te, y_pred_va = cfg.get("rl", {}), 0.5, {"enabled": False, "threshold_method": "fixed_0.5"}, None, None
    if flags["rl"]:
        val_emb = predict_embeddings(model, g_va, device, eval_chunk)
        y_va = g_va.y.numpy()
        t0 = time.perf_counter()
        agent, rl_hist = train_rl_on_validation(val_emb, y_va, model.classifier, cfg, device)
        timing["rl_training_sec"] = time.perf_counter() - t0
        pol = apply_policy(agent, val_emb, y_va, model.classifier, cfg)
        model.set_feature_weights(pol["feature_weights"])      # trọng số RL thực sự đi vào điểm số
        threshold = pol["threshold"]
        rl_info = {"enabled": True, "agent": agent.kind, "num_updates": len(rl_hist["losses"]),
                   "final_loss": float(np.mean(rl_hist["losses"][-100:])) if rl_hist["losses"] else None,
                   "last_episode_reward": rl_hist["episode_rewards"][-1], "policy_threshold": threshold,
                   "threshold_method": f"{agent.kind}_policy_{rl_cfg.get('inference', 'mean')}",
                   "feature_weight_std": float(np.std(pol["feature_weights"]))}
    print(f"[PIPE] threshold={threshold:.4f} ({rl_info['threshold_method']})")

    # ---- 3) Đánh giá (không feedback tiếp) ---------------------------------
    t0 = time.perf_counter()
    val_scores, val_labels = predict_scores(model, g_va, device, timing, eval_chunk)
    test_scores, test_labels = predict_scores(model, g_te, device, timing, eval_chunk)
    if flags["rl"] and rl_cfg.get("inference", "mean") == "per_batch":
        te_emb = predict_embeddings(model, g_te, device, eval_chunk)
        tp = apply_policy(agent, te_emb, test_labels, model.classifier, cfg)   # action chỉ phụ thuộc state (không nhãn)
        y_pred_te = np.zeros(len(test_scores), dtype=bool)
        for (a, b), th in zip(tp["spans"], tp["thresholds"]):
            y_pred_te[a:b] = test_scores[a:b] >= th
    val_m = classification_metrics(val_labels, val_scores, threshold)
    test_m = classification_metrics(test_labels, test_scores, threshold, y_pred=y_pred_te)
    timing["eval_sec"] = time.perf_counter() - t0
    timing["total_runtime_sec"] = time.perf_counter() - t_all

    extra = {}
    deviations = []
    g = cfg.get("graph", {})
    if g.get("max_neighbors_per_node") is not None:
        deviations.append(f"graph.max_neighbors_per_node={g['max_neighbors_per_node']} (cap kỹ thuật, paper không giới hạn)")
    if cfg.get("model", {}).get("temporal_max_seq_len") is not None:
        deviations.append(f"model.temporal_max_seq_len={cfg['model']['temporal_max_seq_len']} (cap kỹ thuật cho GRU)")
    if cfg.get("dataset", {}).get("sample_frac") not in (None, 1.0):
        deviations.append(f"dataset.sample_frac={cfg['dataset']['sample_frac']} (KHÔNG phải paper)")

    print(f"\n[PIPE] TEST AUC-ROC={test_m['auc_roc']:.4f} AUC-PR={test_m['auc_pr']:.4f} F1={test_m['f1']:.4f} "
          f"Recall@1%={test_m['recall_at_1pct']*100:.1f}% FPR={test_m['fpr']:.5f} | total {timing['total_runtime_sec']:.0f}s")
    result = {
        "model_name": "FraudGNN-RL", "flags": flags, "seed": seed,   # Đã thay thế resolve_model_name

def run_pipeline(cfg: Dict[str, Any], return_artifacts: bool = False, save: bool = True):
    df = load_dataset(cfg)
    ds = cfg["dataset"]
    if ds.get("fraud_ratio_pct") is not None:   # thí nghiệm imbalance (Fig. 4)
        df = resample_fraud_ratio(df, float(ds["fraud_ratio_pct"]), ds["label_col"], ds.get("time_col"),
                                  int(ds.get("random_state", 42)))
        print(f"[PIPE] fraud ratio ép về {ds['fraud_ratio_pct']}% -> N={len(df):,}")
    train_df, val_df, test_df = split_dataframe(df, cfg)
    res = run_pipeline_on_split(train_df, val_df, test_df, cfg, return_artifacts=return_artifacts)
    result = res[0] if return_artifacts else res
    if save:
        _save(result, cfg, "metrics")
    return res


def run_cv_pipeline(cfg: Dict[str, Any], n_folds: int = 5) -> Dict[str, Any]:
    """5-fold CV (paper V-A-4)."""
    df = load_dataset(cfg)
    runs = []
    for i, (tr, va, te) in enumerate(get_cv_splits(df, cfg, n_folds), 1):
        print(f"\n[CV] Fold {i}/{n_folds}: train={len(tr):,} val={len(va):,} test={len(te):,}")
        try:
            r = run_pipeline_on_split(tr, va, te, copy.deepcopy(cfg))
            runs.append({"fold": i, **r})
        except Exception as e:  # noqa
            import traceback; traceback.print_exc()
            runs.append({"fold": i, "error": str(e)})
    ok = [r for r in runs if "error" not in r]
    summary = {"n_folds": n_folds, "n_valid": len(ok), "folds": runs}
    for m in ("auc_roc", "auc_pr", "f1", "recall", "precision", "fpr", "recall_at_1pct"):
        v = [r["test_metrics"][m] for r in ok]
        if v:
            summary[f"{m}_mean"], summary[f"{m}_std"] = float(np.mean(v)), float(np.std(v, ddof=1) if len(v) > 1 else 0.0)
    _save(summary, cfg, "cv_results")
    for m in ("auc_roc", "auc_pr", "f1", "recall_at_1pct"):
        if f"{m}_mean" in summary:
            print(f"[CV] {m:<15}: {summary[m+'_mean']:.4f} +/- {summary[m+'_std']:.4f}")
    return summary

# ============================================================
# src/train/train_gnn.py — Huấn luyện TSSGC + classifier (chế độ centralized: "w/o FL").
#
# Paper V-A-4: Adam lr=1e-3, batch size 64, 100 epochs, early stopping patience 10.
#
# [INFERRED]
#  - Loss: BCE với pos_weight = n_neg/n_pos (cắt tại pos_weight_max) để xử lý mất cân bằng.
#  - Early stopping theo AUC-PR trên val (paper không nói tiêu chí).
#  - "batch size 64" không khả thi trên đồ thị giao dịch (mini-batch node cần láng giềng 3-hop):
#      batch_mode="subgraph": mỗi bước = 1 subgraph liên tục theo thời gian gồm `subgraph_nodes` node
#      batch_mode="full"    : toàn đồ thị (1 bước/epoch, có thể lặp `grad_steps_per_epoch` lần)
# ============================================================
from __future__ import annotations

import copy
import time
from typing import Any, Dict, Optional

import numpy as np
import torch
import torch.nn.functional as F

from src.eval.evaluate import iter_node_chunks, predict_scores
from src.eval.metrics import classification_metrics
from src.models.fraudgnn_rl import FraudGNNRL
from src.utils.config import get_flags


def build_model(in_dim: int, cfg: Dict[str, Any], device: str) -> FraudGNNRL:
    m, flags = cfg.get("model", {}), get_flags(cfg)
    model = FraudGNNRL(
        in_dim=in_dim, hidden_dim=int(m.get("hidden_dim", 64)), num_layers=int(m.get("num_layers", 3)),
        num_node_types=int(m.get("num_node_types", 1)), dropout=float(m.get("dropout", 0.2)),
        use_gnn=bool(flags["gnn"]), shared_projection=bool(m.get("shared_projection", True)),
        temporal_max_seq_len=m.get("temporal_max_seq_len"),
    ).to(device)
    return model


def compute_pos_weight(y: torch.Tensor, cap: Optional[float] = 100.0) -> torch.Tensor:
    y = y.flatten()
    pos, neg = (y == 1).sum().float(), (y == 0).sum().float()
    w = (neg / pos.clamp_min(1.0)).clamp_min(1.0)
    return w.clamp_max(cap) if cap else w


def train_steps(model, data, optimizer, pos_w, tcfg: Dict[str, Any], device: str, epochs: int = 1,
                on_epoch_end=None) -> float:
    """Chạy `epochs` epoch trên `data`; trả loss trung bình. Dùng chung cho centralized và FL client."""
    mode = tcfg.get("batch_mode", "full")
    chunk = int(tcfg.get("subgraph_nodes", 4096))
    steps_full = int(tcfg.get("grad_steps_per_epoch", 1))
    clip = float(tcfg.get("grad_clip", 5.0))
    losses = []
    model.train()
    for ep in range(epochs):
        if mode == "subgraph" and data.num_nodes > chunk:
            batches = (data.subgraph(idx) for idx in iter_node_chunks(data.num_nodes, chunk, shuffle=True))
        else:
            batches = (data for _ in range(steps_full))
        for b in batches:
            b = b.to(device)
            optimizer.zero_grad()
            loss = F.binary_cross_entropy_with_logits(model(b), b.y.float(), pos_weight=pos_w)
            loss.backward()
            torch.nn.utils.clip_grad_norm_(model.parameters(), clip)
            optimizer.step()
            losses.append(float(loss.item()))
        if on_epoch_end is not None:
            on_epoch_end(ep)
    return float(np.mean(losses)) if losses else 0.0


def train_tssgc_classifier(train_data, val_data, cfg: Dict[str, Any], device: str,
                           timing: Optional[dict] = None) -> Dict[str, Any]:
    t, flags = cfg.get("train", {}), get_flags(cfg)
    model = build_model(int(train_data.x.size(1)), cfg, device)
    print(f"[GNN] centralized | params={sum(p.numel() for p in model.parameters()):,} | gnn={flags['gnn']}")
    opt = torch.optim.Adam(model.parameters(), lr=float(t.get("learning_rate", 1e-3)),
                           weight_decay=float(t.get("weight_decay", 0.0)))
    pos_w = compute_pos_weight(train_data.y, t.get("pos_weight_max", 100.0)).to(device)
    epochs, patience = int(t.get("epochs", 100)), int(t.get("patience", 10))
    eval_chunk = t.get("eval_chunk_nodes")


    best_state, best, bad, hist = None, -1.0, 0, []
    t0 = time.perf_counter()
    for epoch in range(1, epochs + 1):
        loss = train_steps(model, train_data, opt, pos_w, t, device, epochs=1)
        s, y = predict_scores(model, val_data, device, chunk_nodes=eval_chunk)
        m = classification_metrics(y, s, 0.5)
        hist.append({"epoch": epoch, "train_loss": loss, "val_auc_pr": m["auc_pr"], "val_auc_roc": m["auc_roc"]})
        if epoch == 1 or epoch % 10 == 0:
            print(f"[GNN] ep {epoch:3d} loss={loss:.4f} val AUC-PR={m['auc_pr']:.4f} AUC-ROC={m['auc_roc']:.4f}")
        if m["auc_pr"] > best:
            best, bad, best_state = m["auc_pr"], 0, copy.deepcopy(model.state_dict())
        else:
            bad += 1
            if bad >= patience:
                print(f"[GNN] Early stop @ epoch {epoch} (best val AUC-PR={best:.4f})")
                break
    if best_state is not None:
        model.load_state_dict(best_state)
    total = time.perf_counter() - t0
    if timing is not None:
        timing["gnn_training_sec"] = total
    return {"model": model, "history": hist, "best_val_auc_pr": best, "total_time_sec": total}


__all__ = ["build_model", "compute_pos_weight", "train_steps", "train_tssgc_classifier"]

# ============================================================
# src/train/train_rl.py — Huấn luyện / áp dụng RL agent (NAF mặc định | DQN rời rạc).
#
# [INFERRED] Paper không nói:
#   - RL train trên tập nào  -> VALIDATION (test giữ độc lập; Fig.2 'feedback' dùng nhãn của val).
#   - Threshold cuối cùng    -> trung bình threshold của policy greedy trên val
#                               (rl.inference="mean"), hoặc theo từng batch ("per_batch").
#   - Feature weights cuối   -> trung bình trọng số greedy trên các batch val.
#   - Không update online trên test (online learning là FUTURE WORK của paper).
# ============================================================
from __future__ import annotations

from typing import Any, Dict, Tuple

import numpy as np
import torch

from src.models.dqn_agent import DQNAgent
from src.models.naf_agent import NAFAgent
from src.models.reward import BatchFraudEnv, weights_from_action


def _env(emb: torch.Tensor, labels, classifier, rl: Dict[str, Any]) -> BatchFraudEnv:
    return BatchFraudEnv(emb, labels, classifier, batch_size=int(rl.get("env_batch_size", 256)),
                         fpr_penalty=float(rl.get("fpr_penalty", 1.0)), recall_weight=float(rl.get("recall_weight", 0.25)))


def make_agent(state_dim: int, n_features: int, rl: Dict[str, Any], device: str):
    common = dict(hidden_dim=int(rl.get("hidden_dim", 128)), gamma=float(rl.get("gamma", 0.99)),
                  lr=float(rl.get("learning_rate", 1e-3)), tau=float(rl.get("tau", 0.005)),
                  buffer_size=int(rl.get("buffer_size", 10000)), batch_size=int(rl.get("update_batch_size", 64)),
                  device=device)
    if str(rl.get("type", "naf")).lower() == "dqn":
        return DQNAgent(state_dim, n_features, n_bins=int(rl.get("n_threshold_bins", 19)), **common)
    return NAFAgent(state_dim, n_features, **common)


def train_rl_on_validation(val_emb: torch.Tensor, val_labels: np.ndarray, classifier, cfg: Dict[str, Any],
                           device: str) -> Tuple[Any, Dict[str, Any]]:
    rl = cfg.get("rl", {})
    env = _env(val_emb, val_labels, classifier, rl)
    agent = make_agent(env.state_dim, val_emb.size(1), rl, device)
    epochs = int(rl.get("epochs", 100))
    losses, ep_rewards = [], []
    print(f"[RL] {agent.kind.upper()} | val N={env.n:,} | env_batch={env.bs} | epochs={epochs}")
    for ep in range(epochs):
        s, done, tot = env.reset(), False, 0.0
        while not done:
            a, thr, fw = agent.act(s, explore=True)
            s2, r, done, _ = env.step(thr, weights_from_action(fw))
            agent.remember(s, a, r, s2, float(done))
            l = agent.update()
            if l is not None:
                losses.append(l)
            s, tot = s2, tot + r
        ep_rewards.append(tot)
        if (ep + 1) % max(1, epochs // 5) == 0:
            print(f"[RL] epoch {ep+1:3d}/{epochs} episode_reward={tot:.3f} "
                  f"loss={np.mean(losses[-100:]) if losses else float('nan'):.5f}")
    return agent, {"losses": losses, "episode_rewards": ep_rewards}


@torch.no_grad()
def apply_policy(agent, emb: torch.Tensor, labels, classifier, cfg: Dict[str, Any]) -> Dict[str, Any]:
    """Chạy policy greedy trên (emb, labels) -> thresholds theo batch, trọng số trung bình."""
    env = _env(emb, labels, classifier, cfg.get("rl", {}))
    s, done = env.reset(), False
    thr_list, ws, spans = [], [], []
    while not done:
        _, thr, fw = agent.act(s, explore=False)
        w = weights_from_action(fw)
        s, _, done, info = env.step(thr, w)
        thr_list.append(thr); ws.append(w); spans.append((info["start"], info["end"]))
    w_mean = np.mean(ws, axis=0).astype(np.float32)
    return {"threshold": float(np.mean(thr_list)), "thresholds": thr_list, "spans": spans,
            "feature_weights": w_mean, "num_steps": len(thr_list)}

def load_config(path: str | Path) -> Dict[str, Any]:
    """Load YAML config (UTF-8) và điền flags mặc định."""
    path = Path(path)
    if not path.exists():
        raise FileNotFoundError(f"Config file not found: {path}")
    with path.open("r", encoding="utf-8") as f:
        cfg = yaml.safe_load(f) or {}
    cfg["flags"] = {**DEFAULT_FLAGS, **(cfg.get("flags") or {})}
    return cfg


def deep_update(base: Dict[str, Any], upd: Dict[str, Any]) -> Dict[str, Any]:
    """Trả về bản sao của base được cập nhật đệ quy bởi upd."""
    out = copy.deepcopy(base)
    for k, v in upd.items():
        if isinstance(v, dict) and isinstance(out.get(k), dict):
            out[k] = deep_update(out[k], v)
        else:
            out[k] = copy.deepcopy(v)
    return out


def get_flags(cfg: Dict[str, Any]) -> Dict[str, bool]:
    return {**DEFAULT_FLAGS, **(cfg.get("flags") or {})}


def ensure_dirs(*paths: str | Path) -> None:
    for p in paths:

from __future__ import annotations

import time
import functools
from typing import Any, Callable, Dict, Optional
import psutil
import torch
import warnings


def timer(func: Callable) -> Callable:
    """Decorator để đo thời gian chạy của hàm."""
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        start_time = time.perf_counter()
        result = func(*args, **kwargs)
        elapsed = time.perf_counter() - start_time
        
        if isinstance(result, dict):
            result['_timing'] = result.get('_timing', {})
            result['_timing'][func.__name__] = elapsed
        
        return result
    return wrapper


def measure_latency(model, data, device: str = "cpu", num_runs: int = 100) -> Dict[str, float]:
    """
    Đo latency (thời gian inference) của model.
    
    Args:
        model: PyTorch model
        data: Dữ liệu đầu vào (batch)
        device: 'cpu' hoặc 'cuda'
        num_runs: Số lần chạy để tính trung bình
    
    Returns:
        Dict với các metrics:
        - latency_mean_ms: Trung bình (ms)
        - latency_std_ms: Độ lệch chuẩn (ms)
        - latency_min_ms: Nhỏ nhất (ms)
        - latency_max_ms: Lớn nhất (ms)
        - latency_p50_ms: Percentile 50 (ms)
        - latency_p95_ms: Percentile 95 (ms)
        - latency_p99_ms: Percentile 99 (ms)
        - throughput_per_sec: Throughput (samples/s)
    """
    model.eval()
    data = data.to(device)
    
    # Warmup
    with torch.no_grad():
        for _ in range(10):
            _ = model(data)
    
    if device == "cuda":
        torch.cuda.synchronize()
    
    # Đo latency
    latencies = []
    for _ in range(num_runs):
        start = time.perf_counter()
        with torch.no_grad():
            _ = model(data)
        if device == "cuda":
            torch.cuda.synchronize()
        latencies.append((time.perf_counter() - start) * 1000)  # ms
    
    latencies = sorted(latencies)
    num_samples = data.x.size(0) if hasattr(data, 'x') else 1
    
    return {
        "latency_mean_ms": float(sum(latencies) / len(latencies)),
        "latency_std_ms": float((sum((x - sum(latencies)/len(latencies))**2 for x in latencies) / len(latencies)) ** 0.5),
        "latency_min_ms": float(latencies[0]),
        "latency_max_ms": float(latencies[-1]),
        "latency_p50_ms": float(latencies[int(len(latencies) * 0.5)]),
        "latency_p95_ms": float(latencies[int(len(latencies) * 0.95)]),
        "latency_p99_ms": float(latencies[int(len(latencies) * 0.99)]),
        "throughput_per_sec": float(1000 / (sum(latencies) / len(latencies)) * num_samples),
    }


def get_memory_usage(alert_threshold: float = 0.85, log_memory: bool = True) -> Dict[str, float]:
    """
    Lấy thông tin memory usage và cảnh báo nếu gần OOM.
    
    [OK] TÍCH HỢP VÀO PIPELINE: Tự động log memory trong quá trình chạy
    """
    mem = psutil.virtual_memory()
    
    result = {
        "ram_total_gb": mem.total / (1024**3),
        "ram_available_gb": mem.available / (1024**3),
        "ram_used_gb": mem.used / (1024**3),
        "ram_usage_percent": mem.percent,
    }
    
    # [OK] Log RAM
    if log_memory:
        print(f"[MEMORY] RAM: {result['ram_used_gb']:.2f}GB / {result['ram_total_gb']:.2f}GB ({result['ram_usage_percent']:.1f}%)")
    
    # [OK] Cảnh báo nếu RAM quá cao
    if mem.percent > alert_threshold * 100:
        warnings.warn(
            f"[WARN] High memory usage: {mem.percent:.1f}% "
            f"(threshold: {alert_threshold*100:.0f}%)",
            ResourceWarning
        )
    
    # [OK] Check VRAM (GPU)
    if torch.cuda.is_available():
        result["vram_total_gb"] = torch.cuda.get_device_properties(0).total_memory / (1024**3)
        result["vram_allocated_gb"] = torch.cuda.memory_allocated() / (1024**3)
        result["vram_reserved_gb"] = torch.cuda.memory_reserved() / (1024**3)
        result["vram_max_allocated_gb"] = torch.cuda.max_memory_allocated() / (1024**3)
        result["vram_usage_percent"] = (result["vram_allocated_gb"] / result["vram_total_gb"]) * 100
        
        # [OK] Log VRAM
        if log_memory:
            print(f"[MEMORY] VRAM: {result['vram_allocated_gb']:.2f}GB / {result['vram_total_gb']:.2f}GB ({result['vram_usage_percent']:.1f}%)")
        
        # [OK] Cảnh báo nếu VRAM cao
        if result['vram_usage_percent'] > alert_threshold * 100:
            warnings.warn(
                f"[WARN] High GPU memory: {result['vram_allocated_gb']:.2f}GB / {result['vram_total_gb']:.2f}GB ({result['vram_usage_percent']:.1f}%)",
                ResourceWarning
            )
    
    return result


def log_memory_snapshot(stage: str = "") -> Dict[str, float]:
    """
    [OK] Ghi snapshot memory tại một thời điểm trong pipeline.
    Dùng để debug memory leak.
    
    Args:
        stage: Tên giai đoạn (ví dụ: "before_graph", "after_graph", "before_fl", ...)

    """
    print(f"\n{'='*60}")
    print(f"[CHART] MEMORY SNAPSHOT: {stage if stage else 'Current'}")

    mem_data = get_memory_usage(log_memory=True)
    
    # Thêm disk info
    import shutil
    total, used, free = shutil.disk_usage("/")
    print(f"[MEMORY] Disk Free: {free / (1024**3):.2f} GB")
    
    print(f"{'='*60}\n")
    
    return mem_data


def format_time(seconds: float) -> str:
    """Format seconds thành string dễ đọc."""
    if seconds < 60:
        return f"{seconds:.2f}s"
    elif seconds < 3600:
        return f"{seconds/60:.2f}m"
    else:
        return f"{seconds/3600:.2f}h"


def print_timing_summary(timing: Dict[str, float]):
    """In bảng timing summary đẹp."""
    print("\n" + "="*60)
    print("⏱️ TIMING SUMMARY")
    print("="*60)
    
    # Định nghĩa thứ tự và tên hiển thị
    order = [
        ("data_loading_sec", "Data Loading"),
        ("data_splitting_sec", "Data Splitting"),
        ("preprocessing_sec", "Preprocessing"),
        ("graph_building_sec", "Graph Building"),
        ("federated_training_sec", "Federated Training"),
        ("tssgc_training_sec", "TSSGC Training"),
        ("rl_training_sec", "RL Training"),
        ("inference_sec", "Inference"),
        ("total_runtime_sec", "TOTAL"),
    ]
    
    for key, name in order:
        if key in timing:
            value = timing[key]
            if key == "total_runtime_sec":
                print(f"{name:20} {format_time(value):>15}")
            else:
                print(f"{name:20} {format_time(value):>15}")
